# ~/.bashrc
# Fedora Bootstrap for the Custom-Shell Framework

# 1. Source Fedora Global Definitions
# Loads /etc/profile.d/*.sh (Flatpak paths, VTE terminal integration, system colors, bash-completion)
if [ -f /etc/bashrc ]; then
    . /etc/bashrc
fi

# 2. User Specific Environment PATH
if ! [[ "$PATH" =~ "$HOME/.local/bin:$HOME/bin:" ]]; then
    PATH="$HOME/.local/bin:$HOME/bin:$PATH"
fi
export PATH

# Uncomment the following line if you don't like systemctl's auto-paging feature:
# export SYSTEMD_PAGER=

# 3. User Specific Modular Drop-ins (~/.bashrc.d)
if [ -d ~/.bashrc.d ]; then
    for rc in ~/.bashrc.d/*; do
        if [ -f "$rc" ]; then
            . "$rc"
        fi
    done
fi
unset rc

# 4. Developer Environments & Tooling
# Antigravity CLI
[[ -d "$HOME/.local/bin" ]] && export PATH="$HOME/.local/bin:$PATH"

# NVM (Node Version Manager)
export NVM_DIR="$HOME/.nvm"
[ -s "$NVM_DIR/nvm.sh" ] && \. "$NVM_DIR/nvm.sh"
[ -s "$NVM_DIR/bash_completion" ] && \. "$NVM_DIR/bash_completion"

# FNM (Fast Node Manager)
FNM_PATH="$HOME/.local/share/fnm"
if [ -d "$FNM_PATH" ]; then
    export PATH="$FNM_PATH:$PATH"
    eval "$(fnm env --shell bash)"
fi

# Rust / Cargo
[ -f "$HOME/.cargo/env" ] && . "$HOME/.cargo/env"

# 5. Custom-Shell Framework Entry Point
CUSTOM_SHELL_MAIN="$HOME/Config/custom-shell/main.sh"
if [[ -f "$CUSTOM_SHELL_MAIN" ]]; then
    source "$CUSTOM_SHELL_MAIN"
fi
